ברכת הלבנה

איך ומתי מברכים?

מברכים ברכת הלבנה פעם אחת בחודש, בלילה, כשרואים את הירח בחוץ, תחת כיפת השמים. לא מברכים בתוך בית או תחת גג.

מתחילים לאחר שעברו שבעה ימים מהמולד. בשעת הדחק, ובעיקר בחורף כשהירח מכוסה בעננים, מותר לברך כבר משלושה ימים מהמולד. הזמן נמשך עד חמישה עשר יום מהמולד (לדעת הרמ״א, עד חצי החודש).

מברכים אחרי צאת הכוכבים, כשהירח נראה ומאיר. לכתחילה במוצאי שבת, בבגדים נאים, ובמנין. מי שחושש שיפסיד את הברכה, אין צריך להמתין למוצאי שבת. ברכת חודש אב אחרי תשעה באב, וברכת חודש תשרי במוצאי יום הכיפורים.

עומדים, רגליים צמודות, ומביטים בירח לפני תחילת הברכה. לא מברכים בליל שבת ובליל יום טוב אלא אם נחשש שהזמן יעבור. מי שכבר בירך החודש, אינו מברך שוב. בשאלה מעשית יש לשאול רב.

סֵדֶר בִּרְכַּת הַלְּבָנָה

הַלְלוּיָהּ הַלְלוּ אֶת ה׳ מִן הַשָּׁמַיִם הַלְלוּהוּ בַּמְּרוֹמִים: הַלְלוּהוּ כָל מַלְאָכָיו הַלְלוּהוּ כָּל צְבָאָיו: הַלְלוּהוּ שֶׁמֶשׁ וְיָרֵחַ הַלְלוּהוּ כָּל כּוֹכְבֵי אוֹר: הַלְלוּהוּ שְׁמֵי הַשָּׁמָיִם וְהַמַּיִם אֲשֶׁר מֵעַל הַשָּׁמָיִם: יְהַלְלוּ אֶת שֵׁם ה׳ כִּי הוּא צִוָּה וְנִבְרָאוּ: וַיַּעֲמִידֵם לָעַד לְעוֹלָם חָק נָתַן וְלֹא יַעֲבוֹר:

כשאומר 'כי אראה שמיך' יסתכל בשמים ובלבנה, אבל לכשיתחיל בברכה לא יסתכל בה עוד (בא"ח ויקרא כג).

בָּרוּךְ אַתָּה ה׳, ה׳ מֶלֶךְ הָעוֹלָם, אֲשֶׁר בְּמַאֲמָרוֹ בָּרָא שְׁחָקִים וּבְרוּחַ פִּיו כָּל צְבָאָם, חֹק וּזְמַן נָתַן לָהֶם שֶׁלֹּא יְשַׁנוּ אֶת תַּפְקִידָם, שָׂשִׂים וּשְׂמֵחִים לַעֲשׂוֹת רְצוֹן קוֹנֵיהֶם, פּוֹעֵל אֱמֶת שֶׁפְּעֻלָּתוֹ אֱמֶת, וְלַלְּבָנָה אָמַר שֶׁתִּתְחַדֵּשׁ, עֲטֶרֶת תִּפְאֶרֶת לַעֲמוּסֵי בָטֶן, שֶׁגַּם הֵם עֲתִידִים לְהִתְחַדֵּשׁ כְּמוֹתָהּ וּלְפָאֵר לְיוֹצְרָם עַל שֵׁם כְּבוֹד מַלְכוּתוֹ. בָּרוּךְ אַתָּה ה׳, מְחַדֵּשׁ חֳדָשִׁים:

סִימָן טוֹב תְּהִי לָנוּ וּלְכָל יִשְׂרָאֵל: (ג׳ פעמים)

בָּרוּךְ יוֹצְרֵךְ, בָּרוּךְ עוֹשֵׂךְ, בָּרוּךְ קוֹנֵךְ, בָּרוּךְ בּוֹראֵךְ, כְּשֵׁם שֶׁאֲנַחְנוּ מרַקְּדִים (מדלג ג׳ פעמים) כְּנֶגְדֵּךְ, וְאֵין אֲנַחְנוּ יְכוֹלִים לִגַּע בָּךְ, כָּךְ אִם יְרַקְּדוּ אֲחֵרִים כְּנֶגְדֵּנוּ לְהַזִּיקֵנוּ, לֹא יוּכְלוּ לִגַּע בָּנוּ, וְלֹא יִשְׁלְטוּ בָנוּ, וְלֹא יַעֲשׂוּ בָנוּ שׁוּם רֹשֶׁם כְּלָל ועִקָר. (יש מוסיפים לסגולה: לֹא בגוּפֵנוּ, ולֹא במָמוֹנֵנוּ, ולֹא בּכָל אֲשֶׁר לָנוּ, ולֹא יִהְיוּ לָנוּ כְּאֵבֵי שִׁנָּיִם.) תִּפֹּל עֲלֵיהֶם אֵימָתָה וָפַחַד, בִּגְדֹל זְרוֹעֲךָ יִדְּמוּ כָּאָבֶן: כָּאָבֶן יִדְּמוּ, זְרוֹעֲךָ בִּגְדֹל, וָפַחַד אֵימָתָה עֲלֵיהֶם תִּפֹּל:

דָּוִד מֶלֶךְ יִשְׂרָאֵל חַי וְקַיָּם. (ג׳ פעמים)

אָמֵן, אָמֵן, אָמֵן: נֶצַח, נֶצַח, נֶצַח: סֶלָה, סֶלָה, סֶלָה: וָעֶד, וָעֶד, וָעֶד:

לֵב טָהוֹר בְּרָא לִי ה׳, וְרוּחַ נָכוֹן חַדֵּשׁ בְּקִרְבִּי: (שבע פעמים)

שִׁיר לַמַּעֲלוֹת אֶשָּׂא עֵינַי אֶל הֶהָרִים מֵאַיִן יָבֹא עֶזְרִי: עֶזְרִי מֵעִם ה׳ עֹשֵׂה שָׁמַיִם וָאָרֶץ: אַל יִתֵּן לַמּוֹט רַגְלֶךָ אַל יָנוּם שֹׁמְרֶךָ: הִנֵּה לֹא יָנוּם וְלֹא יִישָׁן שׁוֹמֵר יִשְׂרָאֵל: ה׳ שֹׁמְרֶךָ ה׳ צִלְּךָ עַל יַד יְמִינֶךָ: יוֹמָם הַשֶּׁמֶשׁ לֹא יַכֶּכָּה וְיָרֵחַ בַּלָּיְלָה: ה׳ יִשְׁמָרְךָ מִכָּל רָע יִשְׁמֹר אֶת נַפְשֶׁךָ: ה׳ יִשְׁמָר צֵאתְךָ וּבוֹאֶךָ מֵעַתָּה וְעַד עוֹלָם:

תָּנָא דְבֵי רִבִּי יִשְׁמָעֵאל אִלְמָלֵי לֹא זָכוּ בְנֵי יִשְׂרָאֵל אֶלָּא לְהַקְבִּיל פְּנֵי אֲבִיהֶם שֶׁבַּשָּׁמַיִם פַּעַם אַחַת בַּחֹדֶשׁ דַּיָּם. אָמַר אַבַּיֵּי הֵלְכָךְ נִימְרִינְהוּ מֵעֹמֶד:

שָׁלוֹם עֲלֵיכֶם.

נרשם. התזכורת בעמוד הראשי תיעלם עד החודש הבא.
